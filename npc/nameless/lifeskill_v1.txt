function	script	F_LifeSkillRankName	{
	.@rank = getarg(0);
	if (.@rank <= 1)
		return "Apprentice";
	if (.@rank == 2)
		return "Journeyman";
	return "Artisan";
}

function	script	F_LifeSkillYieldBonus	{
	.@rank = getarg(0);
	.@bonus = 0;
	if (.@rank >= 2)
		.@bonus++;
	if (.@rank >= 3 && rand(100) < 35)
		.@bonus++;
	return .@bonus;
}

function	script	F_LifeSkillGain	{
	.@profession$ = getarg(0);
	if (.@profession$ == "Foraging") {
		ForagingPoints++;
		.@old_rank = ForagingRank;
		if (ForagingRank < 1)
			ForagingRank = 1;
		if (ForagingPoints >= 25)
			ForagingRank = 3;
		else if (ForagingPoints >= 10 && ForagingRank < 2)
			ForagingRank = 2;
		if (ForagingRank > .@old_rank)
			dispbottom "[Life Skill] Your Foraging rank is now " + callfunc("F_LifeSkillRankName", ForagingRank) + ".";
		return ForagingRank;
	}

	MiningPoints++;
	.@old_rank = MiningRank;
	if (MiningRank < 1)
		MiningRank = 1;
	if (MiningPoints >= 25)
		MiningRank = 3;
	else if (MiningPoints >= 10 && MiningRank < 2)
		MiningRank = 2;
	if (MiningRank > .@old_rank)
		dispbottom "[Life Skill] Your Mining rank is now " + callfunc("F_LifeSkillRankName", MiningRank) + ".";
	return MiningRank;
}

		mes "Life Skills split into ^0055FFForaging^000000 and ^AA5500Mining^000000.";
		mes "Foraging feeds your stat food pipeline now.";
		mes "Mining starts here as future gear-crafting infrastructure.";
		close;

	case 2:
		if (!LifeSkillStarterKit) {
			LifeSkillStarterKit = 1;
			ForagingRank = 1;
			MiningRank = 1;
			ForagingPoints = 0;
			MiningPoints = 0;
			getitem 50000,1;
			getitem 50001,1;
			getitem 7481,1;
			mes "You are registered as an Apprentice Forager and Apprentice Miner.";
			mes "I've also issued your basic tools and a Level 10 Cookbook.";
		} else
			mes "You are already registered. I can replace missing tools or manuals.";
		callsub S_GrantFoodUnlocks;
		close;

	case 3:
		mes "Foraging Rank: ^0055FF" + callfunc("F_LifeSkillRankName", ForagingRank) + "^000000";
		mes "Foraging Points: ^0055FF" + ForagingPoints + "^000000";
		mes "Mining Rank: ^AA5500" + callfunc("F_LifeSkillRankName", MiningRank) + "^000000";
		mes "Mining Points: ^AA5500" + MiningPoints + "^000000";
		mes "Unlocked food notes: " + (FoodUnlockStr + FoodUnlockAgi + FoodUnlockDex + FoodUnlockVit + FoodUnlockInt + FoodUnlockLuk) + " / 6";
		close;

	case 4:
		callsub S_GrantFoodUnlocks;
		.@replaced = 0;
		if (LifeSkillStarterKit && countitem(50000) == 0) {
			getitem 50000,1;
			.@replaced++;
		}
		if (LifeSkillStarterKit && countitem(50001) == 0) {
			getitem 50001,1;
			.@replaced++;
		}
		if (LifeSkillStarterKit && countitem(7481) == 0) {
			getitem 7481,1;
			.@replaced++;
		}
		if (FoodUnlockStr && countitem(50015) == 0) {
			getitem 50015,1;
			.@replaced++;
		}
		if (FoodUnlockAgi && countitem(50016) == 0) {
			getitem 50016,1;
			.@replaced++;
		}
		if (FoodUnlockVit && countitem(50017) == 0) {
			getitem 50017,1;
			.@replaced++;
		}
		if (FoodUnlockInt && countitem(50018) == 0) {
			getitem 50018,1;
			.@replaced++;
		}
		if (FoodUnlockDex && countitem(50019) == 0) {
			getitem 50019,1;
			.@replaced++;
		}
		if (FoodUnlockLuk && countitem(50020) == 0) {
			getitem 50020,1;
			.@replaced++;
		}
		if (.@replaced)
			mes "I've replaced your missing toolkit items.";
		else
			mes "You are not missing anything that I can replace.";
		close;

	case 5:
		if (!LifeSkillStarterKit) {
			mes "Register first. Then I can authorize the cooking notes.";
			close;
		}
		callsub S_GrantFoodUnlocks;
		if (countitem(7481) == 0) {
			mes "You need your ^0055FFLevel 10 Cookbook^000000 in inventory.";
			close;
		}
		if (!(FoodUnlockStr || FoodUnlockAgi || FoodUnlockDex || FoodUnlockVit || FoodUnlockInt || FoodUnlockLuk)) {
			mes "You have not unlocked any food notes yet.";
			close;
		}
		mes "I'll open the official cooking interface for your unlocked +10 foods.";
		next;
		cooking 20;
		close;
	}

S_GrantFoodUnlocks:
	if (ForagingRank >= 1 && !FoodUnlockStr) {
		FoodUnlockStr = 1;
		getitem 50015,1;
		mes "Unlocked: ^0055FFSteamed Tongue^000000 notes.";
	}
	if (ForagingRank >= 1 && !FoodUnlockAgi) {
		FoodUnlockAgi = 1;
		getitem 50016,1;
		mes "Unlocked: ^0055FFSteamed Desert Scorpions^000000 notes.";
	}
	if (ForagingRank >= 1 && !FoodUnlockDex) {
		FoodUnlockDex = 1;
		getitem 50019,1;
		mes "Unlocked: ^0055FFHwergelmir's Tonic^000000 notes.";
	}
	if (ForagingRank >= 2 && !FoodUnlockVit) {
		FoodUnlockVit = 1;
		getitem 50017,1;
		mes "Unlocked: ^0055FFImmortal Stew^000000 notes.";
	}
	if (ForagingRank >= 2 && !FoodUnlockInt) {
		FoodUnlockInt = 1;
		getitem 50018,1;
		mes "Unlocked: ^0055FFDragon Breath Cocktail^000000 notes.";
	}
	if (ForagingRank >= 3 && !FoodUnlockLuk) {
		FoodUnlockLuk = 1;
		getitem 50020,1;
		mes "Unlocked: ^0055FFCooked Nine Tail's Tails^000000 notes.";
	}
	return;
}
