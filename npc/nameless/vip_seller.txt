prontera,152,176,6	script	VIP Cash Shop	4_F_KAFRA1,{
	mes "[VIP Cash Shop]";
	mes "Welcome to our premium VIP service!";
	mes "Purchase VIP status with Cash Points.";
	next;
	
	// Check current VIP status
	if (vip_status(VIP_STATUS_ACTIVE)) {
		.@remaining = vip_status(VIP_STATUS_REMAINING);
		.@days = .@remaining / 86400;
		.@hours = (.@remaining % 86400) / 3600;
		.@minutes = ((.@remaining % 86400) % 3600) / 60;
		
		mes "[VIP Cash Shop]";
		mes "^0000FFCurrent VIP Status: ACTIVE^000000";
		mes "Time remaining: ^FF0000" + .@days + "^000000 days, ^FF0000" + .@hours + "^000000 hours, ^FF0000" + .@minutes + "^000000 minutes";
		mes " ";
		mes "You can extend your VIP time below.";
		next;
	} else {
		mes "[VIP Cash Shop]";
		mes "^FF0000Current VIP Status: INACTIVE^000000";
		mes "Purchase VIP to enjoy premium benefits!";
		next;
	}
	
	mes "[VIP Cash Shop]";
	mes "^0000FFVIP Benefits:^000000";
	mes "Increased EXP rates";
	mes "Extra storage slots";
	mes "Double Daily Login rewards";  
	mes "Access to VIP areas";
	next;
	
	mes "[VIP Cash Shop]";
	mes "^0000FFYour Cash Points: " + #CASHPOINTS + "^000000";
	mes " ";
	mes "What would you like to do?";
	next;
	
	switch(select("Purchase VIP for myself:^0000FFGift VIP to another player^000000:VIP Information:Cancel")) {
		case 1: // Purchase for self
			.@gift_mode = 0;
			.@target_name$ = strcharinfo(0);
			break;
			
		case 2: // Gift to another player
			.@gift_mode = 1;
			mes "[VIP Cash Shop]";
			mes "^0000FFGift VIP Service^000000";
			mes " ";
			mes "^FF0000WARNING: Please type the character name correctly!^000000";
			mes "^FF0000There will be NO REFUNDS for wrong names!^000000";
			mes "^FF0000Make sure the player exists before gifting!^000000";
			mes " ";
			mes "Enter the character name to gift VIP to:";
			next;
			input .@target_name$;
			
			// Check if player exists and is online
			.@target_aid = getcharid(3, .@target_name$);
			if (.@target_aid == 0) {
				mes "[VIP Cash Shop]";
				mes "^FF0000ERROR: Character '" + .@target_name$ + "' not found or is offline!^000000";
				mes " ";
				mes "Please make sure:";
				mes "- The character name is spelt correctly";
				mes "- The character exists on this server";
				mes "- The character is currently online";
				close;
			}
			
			// Double confirmation for gifts
			mes "[VIP Cash Shop]";
			mes "^0000FFGift Confirmation^000000";
			mes " ";
			mes "You are about to gift VIP to:";
			mes "^0000FF" + .@target_name$ + "^000000";
			mes " ";
			mes "^FF0000FINAL WARNING:^000000";
			mes "^FF0000Once purchased, this CANNOT be undone!^000000";
			mes "^FF0000NO REFUNDS will be given for any reason!^000000";
			mes " ";
			mes "Are you ABSOLUTELY SURE this is correct?";
			next;
			if (select("^0000FFYes, the name is correct^000000:^FF0000No, let me check again^000000") == 2) {
				mes "[VIP Cash Shop]";
				mes "Purchase cancelled.";
				mes "Please double-check the name and try again.";
				close;
			}
			break;
			
		case 3: // VIP Information
			mes "[VIP Cash Shop]";
			mes "^0000FFVIP Package Comparison:^000000";
			mes " ";
			mes "^FF0000Daily VIP:^000000";
			mes "Cost: 500 Cash Points per day";
			mes "Duration: 1 day";
			mes " ";
			mes "^0000FFWeekly VIP:^000000";
			mes "Cost: 2,800 Cash Points (400 per day)";
			mes "Duration: 7 days";
			mes "Saves 700 Cash Points vs daily!";
			mes " ";
			mes "^FF0000Monthly VIP (RECOMMENDED):^000000";
			mes "Cost: 10,000 Cash Points (333 per day)";
			mes "Duration: 30 days";
			mes "Saves 5,000 Cash Points vs daily!";
			mes "^0000FFBest value for money!^000000";
			close;
			
		case 4: // Cancel
			mes "[VIP Cash Shop]";
			mes "Thank you for considering our VIP service!";
			mes "Come back anytime when you're ready.";
			close;
	}
	
	// Show VIP packages
	mes "[VIP Cash Shop]";
	if (.@gift_mode == 1) {
		mes "^0000FFGifting VIP to: " + .@target_name$ + "^000000";
		mes " ";
	}
	mes "^FF0000VIP Packages:^000000";
	mes "^0000FF1. Daily VIP^000000 - 500 Cash Points (1 day)";
	mes "^0000FF2. Weekly VIP^000000 - 2,800 Cash Points (7 days)";
	mes "   Save 700 Cash Points!";
	mes "^0000FF3. Monthly VIP^000000 - 10,000 Cash Points (30 days)";
	mes "   ^FF0000BEST VALUE! Save 5,000 Cash Points!^000000";
	next;
	
	switch(select("Daily VIP (500 Cash):Weekly VIP (2,800 Cash):Monthly VIP (10,000 Cash):Cancel")) {
		case 1: // Daily VIP
			.@cost = 500;
			.@days = 1;
			.@package$ = "Daily VIP";
			break;
			
		case 2: // Weekly VIP
			.@cost = 2800;
			.@days = 7;
			.@package$ = "Weekly VIP";
			break;
			
		case 3: // Monthly VIP
			.@cost = 10000;
			.@days = 30;
			.@package$ = "Monthly VIP";
			break;
			
		case 4: // Cancel
			mes "[VIP Cash Shop]";
			mes "Purchase cancelled.";
			close;
	}
	
	// Purchase confirmation
	mes "[VIP Cash Shop]";
	if (.@gift_mode == 1) {
		mes "^0000FFGift Purchase Summary:^000000";
		mes "^0000FFRecipient: " + .@target_name$ + "^000000";
		mes " ";
	}
	mes "^0000FFPackage: " + .@package$ + "^000000";
	mes "^FF0000Cost: " + .@cost + " Cash Points^000000";
	mes "^0000FFDuration: " + .@days + " day" + (.@days > 1 ? "s" : "") + "^000000";
	mes " ";
	
	if (#CASHPOINTS < .@cost) {
		mes "^FF0000Sorry, you don't have enough Cash Points!^000000";
		mes "Required: " + .@cost + " Cash Points";
		mes "You have: " + #CASHPOINTS + " Cash Points";
		mes "You need: " + (.@cost - #CASHPOINTS) + " more Cash Points";
		close;
	}
	
	if (.@gift_mode == 1) {
		mes "^FF0000FINAL CONFIRMATION:^000000";
		mes "^FF0000NO REFUNDS after purchase!^000000";
		mes " ";
	}
	mes "Are you sure you want to purchase this VIP package?";
	next;
	
	if (select("Yes, purchase VIP:No, cancel") == 2) {
		mes "[VIP Cash Shop]";
		mes "Purchase cancelled.";
		close;
	}
	
	// Process purchase
	#CASHPOINTS -= .@cost;
	
	if (.@gift_mode == 1) {
		// Gift to another player
		vip_time (.@days * 24 * 60), .@target_name$;
		
		mes "[VIP Cash Shop]";
		mes "^0000FFGift purchase successful!^000000";
		mes " ";
		mes "^0000FFVIP gifted to: " + .@target_name$ + "^000000";
		mes "^0000FFPackage: " + .@package$ + "^000000";
		mes "^FF0000Cash Points spent: " + .@cost + "^000000";
		mes "^0000FFRemaining Cash Points: " + #CASHPOINTS + "^000000";
		mes " ";
		mes "^FF0000" + .@days + " day" + (.@days > 1 ? "s" : "") + " of VIP has been gifted!^000000";
		
		// Notify the recipient
		if (isloggedin(.@target_aid)) {
			message .@target_name$, "You have received a VIP gift! " + .@package$ + " (" + .@days + " day" + (.@days > 1 ? "s" : "") + ") from " + strcharinfo(0) + "!";
		}
	} else {
		// Purchase for self
		vip_time (.@days * 24 * 60);
		
		mes "[VIP Cash Shop]";
		mes "^0000FFPurchase successful!^000000";
		mes " ";
		mes "^0000FFPackage purchased: " + .@package$ + "^000000";
		mes "^FF0000Cash Points spent: " + .@cost + "^000000";
		mes "^0000FFRemaining Cash Points: " + #CASHPOINTS + "^000000";
		mes " ";
		mes "^FF0000" + .@days + " day" + (.@days > 1 ? "s" : "") + " of VIP has been added to your account!^000000";
		
		// Show new VIP status
		.@remaining = vip_status(VIP_STATUS_REMAINING);
		.@total_days = .@remaining / 86400;
		.@total_hours = (.@remaining % 86400) / 3600;
		
		mes " ";
		mes "^0000FFNew VIP Status:^000000";
		mes "Total VIP time: ^FF0000" + .@total_days + "^000000 days, ^FF0000" + .@total_hours + "^000000 hours";
	}
	
	next;
	mes "[VIP Cash Shop]";
	mes "Thank you for your purchase!";
	if (.@gift_mode == 1) {
		mes "Your gift has been delivered!";
	} else {
		mes "Enjoy your VIP benefits!";
	}
	mes " ";
	mes "^0000FFPro Tip:^000000 Monthly VIP offers the best value!";
	mes "Save up to 5,000 Cash Points compared to daily purchases.";
	close;
}
