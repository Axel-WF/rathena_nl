prontera,148,183,2	script	Master Craftsman	73,{
	disable_items;

	mes "[Master Craftsman]";
	mes "I enchant ^0055FFbody armor^000000 one piece at a time.";
	mes "I keep your refine, cards in slots 1 to 3, bound state, and random options.";
	mes "^FF5555Slot 4 will be overwritten^000000, even if the armor does not normally expose it.";
	if (.ZenyFee > 0)
		mes "Service fee: ^FFAA00" + .ZenyFee + " Zeny^000000";
	next;

	deletearray .@armor_idx[0], 100;
	deletearray .@armor_id[0], 100;
	deletearray .@armor_refine[0], 100;
	deletearray .@armor_identify[0], 100;
	deletearray .@armor_attribute[0], 100;
	deletearray .@armor_bound[0], 100;
	deletearray .@armor_card1[0], 100;
	deletearray .@armor_card2[0], 100;
	deletearray .@armor_card3[0], 100;
	deletearray .@armor_card4[0], 100;
	deletearray .@armor_opt_id1[0], 100;
	deletearray .@armor_opt_id2[0], 100;
	deletearray .@armor_opt_id3[0], 100;
	deletearray .@armor_opt_id4[0], 100;
	deletearray .@armor_opt_id5[0], 100;
	deletearray .@armor_opt_val1[0], 100;
	deletearray .@armor_opt_val2[0], 100;
	deletearray .@armor_opt_val3[0], 100;
	deletearray .@armor_opt_val4[0], 100;
	deletearray .@armor_opt_val5[0], 100;
	deletearray .@armor_opt_param1[0], 100;
	deletearray .@armor_opt_param2[0], 100;
	deletearray .@armor_opt_param3[0], 100;
	deletearray .@armor_opt_param4[0], 100;
	deletearray .@armor_opt_param5[0], 100;
	.@armor_count = 0;
	.@menu$ = "";

	getinventorylist();
	freeloop(1);
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] <= 0)
			continue;
		if (@inventorylist_equip[.@i])
			continue;
		if (getiteminfo(@inventorylist_id[.@i], ITEMINFO_TYPE) != IT_ARMOR)
			continue;
		if (!(getiteminfo(@inventorylist_id[.@i], ITEMINFO_LOCATIONS) & EQP_ARMOR))
			continue;

		if (.@armor_count >= 100) {
			freeloop(0);
			mes "[Master Craftsman]";
			mes "You have too many candidate armors in inventory.";
			mes "Clean it up a little and try again.";
			close;
		}

		.@armor_idx[.@armor_count] = @inventorylist_idx[.@i];
		.@armor_id[.@armor_count] = @inventorylist_id[.@i];
		.@armor_refine[.@armor_count] = @inventorylist_refine[.@i];
		.@armor_identify[.@armor_count] = @inventorylist_identify[.@i];
		.@armor_attribute[.@armor_count] = @inventorylist_attribute[.@i];
		.@armor_bound[.@armor_count] = @inventorylist_bound[.@i];
		.@armor_card1[.@armor_count] = @inventorylist_card1[.@i];
		.@armor_card2[.@armor_count] = @inventorylist_card2[.@i];
		.@armor_card3[.@armor_count] = @inventorylist_card3[.@i];
		.@armor_card4[.@armor_count] = @inventorylist_card4[.@i];
		.@armor_opt_id1[.@armor_count] = @inventorylist_option_id1[.@i];
		.@armor_opt_id2[.@armor_count] = @inventorylist_option_id2[.@i];
		.@armor_opt_id3[.@armor_count] = @inventorylist_option_id3[.@i];
		.@armor_opt_id4[.@armor_count] = @inventorylist_option_id4[.@i];
		.@armor_opt_id5[.@armor_count] = @inventorylist_option_id5[.@i];
		.@armor_opt_val1[.@armor_count] = @inventorylist_option_value1[.@i];
		.@armor_opt_val2[.@armor_count] = @inventorylist_option_value2[.@i];
		.@armor_opt_val3[.@armor_count] = @inventorylist_option_value3[.@i];
		.@armor_opt_val4[.@armor_count] = @inventorylist_option_value4[.@i];
		.@armor_opt_val5[.@armor_count] = @inventorylist_option_value5[.@i];
		.@armor_opt_param1[.@armor_count] = @inventorylist_option_parameter1[.@i];
		.@armor_opt_param2[.@armor_count] = @inventorylist_option_parameter2[.@i];
		.@armor_opt_param3[.@armor_count] = @inventorylist_option_parameter3[.@i];
		.@armor_opt_param4[.@armor_count] = @inventorylist_option_parameter4[.@i];
		.@armor_opt_param5[.@armor_count] = @inventorylist_option_parameter5[.@i];

		.@menu$ += "#" + (.@armor_count + 1) + " " + getitemname(@inventorylist_id[.@i]) + " [+" + @inventorylist_refine[.@i] + "]:";
		.@armor_count++;
	}
	freeloop(0);

	if (!.@armor_count) {
		mes "[Master Craftsman]";
		mes "You do not have any ^0055FFunequipped body armor^000000 I can work on.";
		close;
	}

	.@menu$ += "Cancel";
	mes "[Master Craftsman]";
	mes "Choose the armor you want me to enchant.";
	next;

	.@selected = select(.@menu$) - 1;
	if (.@selected == .@armor_count) {
		mes "[Master Craftsman]";
		mes "Come back when you have something worth improving.";
		close;
	}

	deletearray .@tier_map[0], 3;
	.@tier_count = 0;
	.@tier_menu$ = "";
	for (.@i = 0; .@i < 3; .@i++) {
		.@mat_id = .TierMaterial[.@i];
		.@mat_count = countitem(.@mat_id);
		if (.@mat_count < 1)
			continue;

		.@tier_map[.@tier_count] = .@i + 1;
		if (.@i == 0)
			.@tier_menu$ += getitemname(.@mat_id) + " x" + .@mat_count + ":";
		else if (.@i == 1)
			.@tier_menu$ += getitemname(.@mat_id) + " x" + .@mat_count + ":";
		else
			.@tier_menu$ += getitemname(.@mat_id) + " x" + .@mat_count + ":";
		.@tier_count++;
	}

	if (!.@tier_count) {
		mes "[Master Craftsman]";
		mes "You do not have any of my current tier materials.";
		mes "Placeholder materials are:";
		mes getitemname(.TierMaterial[0]) + ", " + getitemname(.TierMaterial[1]) + ", and " + getitemname(.TierMaterial[2]) + ".";
		close;
	}

	.@tier_menu$ += "Cancel";
	mes "[Master Craftsman]";
	mes "Now choose the material tier.";
	next;

	.@tier_choice = select(.@tier_menu$) - 1;
	if (.@tier_choice == .@tier_count) {
		mes "[Master Craftsman]";
		mes "No enchant today, then.";
		close;
	}

	.@tier = .@tier_map[.@tier_choice];
	.@material_id = .TierMaterial[.@tier - 1];
	.@selected_item_id = .@armor_id[.@selected];
	.@current_slot4 = .@armor_card4[.@selected];

	mes "[Master Craftsman]";
	mes "Armor: ^0000FF" + getitemname(.@selected_item_id) + "^000000 [+" + .@armor_refine[.@selected] + "]";
	mes "Tier material: ^FFAA00" + getitemname(.@material_id) + "^000000";
	if (.@current_slot4 > 0)
		mes "^FF5555Current slot 4 value will be overwritten:^000000 " + getitemname(.@current_slot4);
	else
		mes "Slot 4 is currently empty.";

	if (.@tier == 1)
		mes "Result rates: 60% +1, 35% +2, 5% +3.";
	else if (.@tier == 2)
		mes "Result rates: 30% +1, 60% +2, 10% +3.";
	else
		mes "Result rates: 10% +1, 20% +2, 70% +3.";

	mes "The stat itself is random among STR, AGI, VIT, INT, DEX, and LUK.";
	next;

	if (.ZenyFee > 0 && Zeny < .ZenyFee) {
		mes "[Master Craftsman]";
		mes "You do not have enough Zeny.";
		close;
	}
	if (countitem(.@material_id) < 1) {
		mes "[Master Craftsman]";
		mes "You no longer have the required material.";
		close;
	}

	mes "[Master Craftsman]";
	mes "Proceed with the enchant?";
	if (select("Enchant it:Cancel") == 2) {
		mes "[Master Craftsman]";
		mes "Maybe next time.";
		close;
	}

	close2;
	progressbar "66ccff", 2;
	specialeffect2 EF_MAPPILLAR;

	delitem .@material_id, 1;
	if (!delitemidx(.@armor_idx[.@selected], 1)) {
		getitem .@material_id, 1;
		mes "[Master Craftsman]";
		mes "Your inventory changed while I was working.";
		mes "I refunded the material. Try again.";
		close;
	}

	if (.ZenyFee > 0)
		Zeny -= .ZenyFee;

	.@bonus_level = callsub(L_RollBonusLevel, .@tier);
	.@stat_index = rand(0, 5);
	.@enchant_item_id = 4700 + (.@stat_index * 10) + (.@bonus_level - 1);

	setarray .@opt_id[0], .@armor_opt_id1[.@selected], .@armor_opt_id2[.@selected], .@armor_opt_id3[.@selected], .@armor_opt_id4[.@selected], .@armor_opt_id5[.@selected];
	setarray .@opt_value[0], .@armor_opt_val1[.@selected], .@armor_opt_val2[.@selected], .@armor_opt_val3[.@selected], .@armor_opt_val4[.@selected], .@armor_opt_val5[.@selected];
	setarray .@opt_param[0], .@armor_opt_param1[.@selected], .@armor_opt_param2[.@selected], .@armor_opt_param3[.@selected], .@armor_opt_param4[.@selected], .@armor_opt_param5[.@selected];

	.@has_options = 0;
	for (.@i = 0; .@i < 5; .@i++) {
		if (.@opt_id[.@i]) {
			.@has_options = 1;
			break;
		}
	}

	if (.@armor_bound[.@selected] > 0) {
		if (.@has_options)
			getitembound3 .@selected_item_id, 1, .@armor_identify[.@selected], .@armor_refine[.@selected], .@armor_attribute[.@selected], .@armor_card1[.@selected], .@armor_card2[.@selected], .@armor_card3[.@selected], .@enchant_item_id, .@armor_bound[.@selected], .@opt_id, .@opt_value, .@opt_param;
		else
			getitembound2 .@selected_item_id, 1, .@armor_identify[.@selected], .@armor_refine[.@selected], .@armor_attribute[.@selected], .@armor_card1[.@selected], .@armor_card2[.@selected], .@armor_card3[.@selected], .@enchant_item_id, .@armor_bound[.@selected];
	} else {
		if (.@has_options)
			getitem3 .@selected_item_id, 1, .@armor_identify[.@selected], .@armor_refine[.@selected], .@armor_attribute[.@selected], .@armor_card1[.@selected], .@armor_card2[.@selected], .@armor_card3[.@selected], .@enchant_item_id, .@opt_id, .@opt_value, .@opt_param;
		else
			getitem2 .@selected_item_id, 1, .@armor_identify[.@selected], .@armor_refine[.@selected], .@armor_attribute[.@selected], .@armor_card1[.@selected], .@armor_card2[.@selected], .@armor_card3[.@selected], .@enchant_item_id;
	}

	dispbottom "Armor enchant complete: " + getitemname(.@selected_item_id) + " received " + getitemname(.@enchant_item_id) + ".";

	mes "[Master Craftsman]";
	mes "Done.";
	mes "Result enchant: ^00AA00" + getitemname(.@enchant_item_id) + "^000000";
	mes "Slot 4 has been updated.";
	close;

L_RollBonusLevel:
	.@tier = getarg(0);
	.@roll = rand(1, 100);

	if (.@tier == 1) {
		if (.@roll <= 60) return 1;
		if (.@roll <= 95) return 2;
		return 3;
	}
	if (.@tier == 2) {
		if (.@roll <= 30) return 1;
		if (.@roll <= 90) return 2;
		return 3;
	}

	if (.@roll <= 10) return 1;
	if (.@roll <= 30) return 2;
	return 3;

OnInit:
	.ZenyFee = 500000;
	setarray .TierMaterial[0], 35600, 35601, 35602; // Tier 1, Tier 2, Tier 3
	end;
}

prontera,148,186,4	script	Essence Fuser	100,{
	mes "[Essence Fuser]";
	mes "I combine essences upward at a ^0055FF3 to 1^000000 rate.";
	mes "Three Tier 1 become one Tier 2, and three Tier 2 become one Tier 3.";
	next;

	switch (select("3x T1 -> 1x T2:3x T2 -> 1x T3:Cancel")) {
	case 1:
		.@from_id = 35600;
		.@to_id = 35601;
		.@fee = 30000;
		break;
	case 2:
		.@from_id = 35601;
		.@to_id = 35602;
		.@fee = 100000;
		break;
	default:
		mes "[Essence Fuser]";
		mes "Come back when you have extra essence to condense.";
		close;
	}

	.@from_name$ = getitemname(.@from_id);
	.@to_name$ = getitemname(.@to_id);
	.@max_by_items = countitem(.@from_id) / 3;
	.@max_by_zeny = Zeny / .@fee;
	.@max_fusions = .@max_by_items;
	if (.@max_by_zeny < .@max_fusions)
		.@max_fusions = .@max_by_zeny;

	if (countitem(.@from_id) < 3) {
		mes "[Essence Fuser]";
		mes "You need at least ^FF55553 " + .@from_name$ + "^000000.";
		close;
	}
	if (Zeny < .@fee) {
		mes "[Essence Fuser]";
		mes "You need ^FF5555" + .@fee + " Zeny^000000 for this fusion.";
		close;
	}

	mes "[Essence Fuser]";
	mes "I will turn ^0055FF3 " + .@from_name$ + "^000000 into ^00AA001 " + .@to_name$ + "^000000.";
	mes "Fee: ^FFAA00" + .@fee + " Zeny^000000";
	mes "You can currently perform ^00AA00" + .@max_fusions + "^000000 fusion(s).";
	switch (select("Fuse 1:Fuse all possible:Cancel")) {
	case 1:
		.@fusion_count = 1;
		break;
	case 2:
		.@fusion_count = .@max_fusions;
		break;
	default:
		mes "[Essence Fuser]";
		mes "No fusion this time.";
		close;
	}

	.@required_items = .@fusion_count * 3;
	.@required_zeny = .@fusion_count * .@fee;

	delitem .@from_id, .@required_items;
	Zeny -= .@required_zeny;
	getitem .@to_id, .@fusion_count;
	specialeffect2 EF_MAPPILLAR;

	mes "[Essence Fuser]";
	mes "Done.";
	mes "You received ^00AA00" + .@fusion_count + " " + .@to_name$ + "^000000.";
	close;
}
