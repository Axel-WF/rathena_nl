prontera,153,183,6	script	Refiner Pro Max	642,{

	mes "[Bulk Refiner]";
	mes "I only process ^0055FFunequipped^000000 equipment.";
	mes "Choose an item and I will refine each eligible copy one by one.";
	next;

	.@refinable_count = 0;
	deletearray .@unique_items[0], 128;
	deletearray .@unique_counts[0], 128;

	getinventorylist();
	freeloop(1);
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] <= 0)
			continue;
		if (@inventorylist_equip[.@i])
			continue;

		.@item_id = @inventorylist_id[.@i];
		.@refine_level = @inventorylist_refine[.@i];

		if (.@refine_level >= 10)
			continue;

		.@item_type = getiteminfo(.@item_id, ITEMINFO_TYPE);
		.@item_location = getiteminfo(.@item_id, ITEMINFO_LOCATIONS);
		.@refineable = 0;
		query_sql(
			"SELECT COALESCE(" +
				"(SELECT `refineable` FROM `item_db2` WHERE `id` = " + .@item_id + " LIMIT 1)," +
				"(SELECT `refineable` FROM `item_db` WHERE `id` = " + .@item_id + " LIMIT 1)," +
				"0)",
			.@refineable
		);

		if (!.@refineable)
			continue;

		.@is_refinable = 0;
		if (.@item_type == IT_WEAPON) {
			.@is_refinable = 1;
		} else if (.@item_type == IT_ARMOR) {
			if ((.@item_location & EQP_ARMOR) ||
				(.@item_location & EQP_HEAD_TOP) ||
				(.@item_location & EQP_HEAD_MID) ||
				(.@item_location & EQP_HEAD_LOW) ||
				(.@item_location & EQP_SHOES) ||
				(.@item_location & EQP_GARMENT) ||
				(.@item_location & EQP_HAND_L)) {
				.@is_refinable = 1;
			}
		}

		if (!.@is_refinable)
			continue;

		.@already_added = 0;
		for (.@k = 0; .@k < .@refinable_count; .@k++) {
			if (.@unique_items[.@k] == .@item_id) {
				.@unique_counts[.@k] += @inventorylist_amount[.@i];
				.@already_added = 1;
				break;
			}
		}

		if (!.@already_added) {
			.@unique_items[.@refinable_count] = .@item_id;
			.@unique_counts[.@refinable_count] = @inventorylist_amount[.@i];
			.@refinable_count++;
		}
	}
	freeloop(0);

	if (!.@refinable_count) {
		mes "[Bulk Refiner]";
		mes "You do not have any ^0055FFunequipped^000000 refinable items below +10.";
		close;
	}

	.@menu_options$ = "";
	for (.@i = 0; .@i < .@refinable_count; .@i++) {
		.@menu_options$ += getitemname(.@unique_items[.@i]) + " (" + .@unique_counts[.@i] + "):";
	}
	.@menu_options$ += "Cancel";

	mes "[Bulk Refiner]";
	mes "I found ^00AA00" + .@refinable_count + "^000000 eligible item types.";
	mes "Which one should I process?";
	next;

	.@selected = select(.@menu_options$) - 1;
	if (.@selected == .@refinable_count) {
		mes "[Bulk Refiner]";
		mes "Come back when you are ready.";
		close;
	}

	.@selected_item_id = .@unique_items[.@selected];
	.@selected_name$ = getitemname(.@selected_item_id);
	.@item_type = getiteminfo(.@selected_item_id, ITEMINFO_TYPE);

	.@material_id = 0;
	.@material_name$ = "";
	.@base_fee = 0;
	.@weapon_level = 0;

	if (.@item_type == IT_WEAPON) {
		.@weapon_level = getiteminfo(.@selected_item_id, ITEMINFO_WEAPONLEVEL);

		if (.@weapon_level == 1) {
			.@material_id = 1010;
			.@material_name$ = "Phracon";
			.@base_fee = 50;
		} else if (.@weapon_level == 2) {
			.@material_id = 1011;
			.@material_name$ = "Emveretarcon";
			.@base_fee = 200;
		} else if (.@weapon_level == 3) {
			.@material_id = 984;
			.@material_name$ = "Oridecon";
			.@base_fee = 5000;
		} else {
			.@material_id = 984;
			.@material_name$ = "Oridecon";
			.@base_fee = 20000;
		}
	} else if (.@item_type == IT_ARMOR) {
		.@material_id = 985;
		.@material_name$ = "Elunium";
		.@base_fee = 2000;
	} else {
		mes "[Bulk Refiner]";
		mes "This item type is not supported.";
		close;
	}

	deletearray .@refine_counts[0], 11;
	.@total_items = 0;

	getinventorylist();
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] != .@selected_item_id)
			continue;
		if (@inventorylist_equip[.@i])
			continue;

		.@current_refine = @inventorylist_refine[.@i];
		if (.@current_refine > 10)
			.@current_refine = 10;

		.@refine_counts[.@current_refine] += @inventorylist_amount[.@i];
		.@total_items += @inventorylist_amount[.@i];
	}

	mes "[Bulk Refiner]";
	mes "Selected item: ^0000FF" + .@selected_name$ + "^000000";
	mes "Eligible unequipped copies: ^00AA00" + .@total_items + "^000000";
	mes "Material per attempt: ^FF6600" + .@material_name$ + "^000000";
	mes "Zeny per attempt: ^FF6600" + .@base_fee + "^000000";
	next;

	dispbottom "=== " + .@selected_name$ + " Breakdown ===";
	for (.@refine = 0; .@refine <= 10; .@refine++) {
		if (.@refine_counts[.@refine] > 0)
			dispbottom "+" + .@refine + ": " + .@refine_counts[.@refine] + " item(s)";
	}

	mes "[Bulk Refiner]";
	mes "What target refine level do you want?";
	mes "^FF0000Warning:^000000 failed attempts destroy the item.";
	input .@target_refine, 1, 10;

	if (.@target_refine < 1 || .@target_refine > 10) {
		mes "[Bulk Refiner]";
		mes "The target refine level must be between 1 and 10.";
		close;
	}

	.@total_zeny_cost = 0;
	.@total_material_cost = 0;
	.@processable_items = 0;

	for (.@refine = 0; .@refine < .@target_refine; .@refine++) {
		if (!.@refine_counts[.@refine])
			continue;

		.@items_at_level = .@refine_counts[.@refine];
		.@processable_items += .@items_at_level;

		for (.@level = .@refine; .@level < .@target_refine; .@level++) {
			.@total_zeny_cost += (.@items_at_level * .@base_fee);
			.@total_material_cost += .@items_at_level;
		}
	}

	if (!.@processable_items) {
		mes "[Bulk Refiner]";
		mes "All eligible copies are already at or above your target.";
		close;
	}

	mes "[Bulk Refiner]";
	mes "Target refine: ^00AA00+" + .@target_refine + "^000000";
	mes "Items to process: ^00AA00" + .@processable_items + "^000000";
	mes "Untouched copies already at target or higher: ^808080" + (.@total_items - .@processable_items) + "^000000";
	mes " ";
	mes "^FF6600Worst-case cost:^000000";
	mes .@total_material_cost + " " + .@material_name$;
	mes .@total_zeny_cost + " Zeny";
	next;

	if (countitem(.@material_id) < .@total_material_cost) {
		mes "[Bulk Refiner]";
		mes "You do not have enough materials.";
		mes "Required: " + .@total_material_cost + " " + .@material_name$;
		mes "You have: " + countitem(.@material_id) + " " + .@material_name$;
		close;
	}

	if (Zeny < .@total_zeny_cost) {
		mes "[Bulk Refiner]";
		mes "You do not have enough Zeny.";
		mes "Required: " + .@total_zeny_cost;
		mes "You have: " + Zeny;
		close;
	}

	mes "[Bulk Refiner]";
	mes "Proceed with bulk refining?";
	if (select("Yes, refine them:No, cancel") == 2) {
		mes "[Bulk Refiner]";
		mes "Cancelled.";
		close;
	}

	deletearray .@process_idx[0], 100;
	deletearray .@process_refine[0], 100;
	deletearray .@process_identify[0], 100;
	deletearray .@process_attribute[0], 100;
	deletearray .@process_bound[0], 100;
	deletearray .@process_card1[0], 100;
	deletearray .@process_card2[0], 100;
	deletearray .@process_card3[0], 100;
	deletearray .@process_card4[0], 100;
	deletearray .@process_opt_id1[0], 100;
	deletearray .@process_opt_id2[0], 100;
	deletearray .@process_opt_id3[0], 100;
	deletearray .@process_opt_id4[0], 100;
	deletearray .@process_opt_id5[0], 100;
	deletearray .@process_opt_val1[0], 100;
	deletearray .@process_opt_val2[0], 100;
	deletearray .@process_opt_val3[0], 100;
	deletearray .@process_opt_val4[0], 100;
	deletearray .@process_opt_val5[0], 100;
	deletearray .@process_opt_param1[0], 100;
	deletearray .@process_opt_param2[0], 100;
	deletearray .@process_opt_param3[0], 100;
	deletearray .@process_opt_param4[0], 100;
	deletearray .@process_opt_param5[0], 100;
	.@process_count = 0;

	getinventorylist();
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] != .@selected_item_id)
			continue;
		if (@inventorylist_equip[.@i])
			continue;
		if (@inventorylist_refine[.@i] >= .@target_refine)
			continue;

		if (.@process_count >= 100) {
			mes "[Bulk Refiner]";
			mes "Safety limit reached. I only process 100 items per batch.";
			close;
		}

		.@process_idx[.@process_count] = @inventorylist_idx[.@i];
		.@process_refine[.@process_count] = @inventorylist_refine[.@i];
		.@process_identify[.@process_count] = @inventorylist_identify[.@i];
		.@process_attribute[.@process_count] = @inventorylist_attribute[.@i];
		.@process_bound[.@process_count] = @inventorylist_bound[.@i];
		.@process_card1[.@process_count] = @inventorylist_card1[.@i];
		.@process_card2[.@process_count] = @inventorylist_card2[.@i];
		.@process_card3[.@process_count] = @inventorylist_card3[.@i];
		.@process_card4[.@process_count] = @inventorylist_card4[.@i];
		.@process_opt_id1[.@process_count] = @inventorylist_option_id1[.@i];
		.@process_opt_id2[.@process_count] = @inventorylist_option_id2[.@i];
		.@process_opt_id3[.@process_count] = @inventorylist_option_id3[.@i];
		.@process_opt_id4[.@process_count] = @inventorylist_option_id4[.@i];
		.@process_opt_id5[.@process_count] = @inventorylist_option_id5[.@i];
		.@process_opt_val1[.@process_count] = @inventorylist_option_value1[.@i];
		.@process_opt_val2[.@process_count] = @inventorylist_option_value2[.@i];
		.@process_opt_val3[.@process_count] = @inventorylist_option_value3[.@i];
		.@process_opt_val4[.@process_count] = @inventorylist_option_value4[.@i];
		.@process_opt_val5[.@process_count] = @inventorylist_option_value5[.@i];
		.@process_opt_param1[.@process_count] = @inventorylist_option_parameter1[.@i];
		.@process_opt_param2[.@process_count] = @inventorylist_option_parameter2[.@i];
		.@process_opt_param3[.@process_count] = @inventorylist_option_parameter3[.@i];
		.@process_opt_param4[.@process_count] = @inventorylist_option_parameter4[.@i];
		.@process_opt_param5[.@process_count] = @inventorylist_option_parameter5[.@i];
		.@process_count++;
	}

	if (!.@process_count) {
		mes "[Bulk Refiner]";
		mes "There is nothing left to process. Your inventory changed while the dialog was open.";
		close;
	}

	.@processed_items = 0;
	.@returned_items = 0;
	.@broken_items = 0;
	.@missing_items = 0;
	.@stopped_items = 0;
	.@successful_refines = 0;
	.@failed_refines = 0;
	.@total_zeny_spent = 0;
	.@total_materials_used = 0;
	.@resources_depleted = 0;
	.@attempt_delay = 200;

	dispbottom "=== Refiner Pro Max starting ===";

	freeloop(1);
	for (.@i = 0; .@i < .@process_count; .@i++) {
		if (countitem(.@material_id) < 1 || Zeny < .@base_fee) {
			.@stopped_items += (.@process_count - .@i);
			break;
		}

		if (!delitemidx(.@process_idx[.@i], 1)) {
			.@missing_items++;
			continue;
		}

		.@processed_items++;
		.@current_refine = .@process_refine[.@i];
		.@item_broken = 0;
		.@stopped_early = 0;

		dispbottom "Processing " + (.@i + 1) + "/" + .@process_count + ": " + .@selected_name$ + " [+" + .@current_refine + "]";

		while (.@current_refine < .@target_refine && !.@item_broken) {
			if (countitem(.@material_id) < 1 || Zeny < .@base_fee) {
				.@stopped_early = 1;
				.@resources_depleted = 1;
				break;
			}

			delitem .@material_id, 1;
			Zeny -= .@base_fee;
			.@total_materials_used++;
			.@total_zeny_spent += .@base_fee;

			.@success_rate = callsub(L_GetSuccessRate, .@item_type, .@weapon_level, .@current_refine);
			.@roll = rand(1, 100);

			if (.@roll <= .@success_rate) {
				.@previous_refine = .@current_refine;
				.@current_refine++;
				.@successful_refines++;
				dispbottom "  -> Success! [+" + .@previous_refine + "] -> [+" + .@current_refine + "] (Roll: " + .@roll + "/" + .@success_rate + "%)";
			} else {
				.@failed_refines++;
				.@item_broken = 1;
				.@broken_items++;
				dispbottom "  -> BROKE! [+" + .@current_refine + "] destroyed! (Roll: " + .@roll + "/" + .@success_rate + "%)";
			}

			sleep2 .@attempt_delay;
		}

		if (!.@item_broken) {
			setarray .@opt_id[0],
				.@process_opt_id1[.@i], .@process_opt_id2[.@i], .@process_opt_id3[.@i], .@process_opt_id4[.@i], .@process_opt_id5[.@i];
			setarray .@opt_value[0],
				.@process_opt_val1[.@i], .@process_opt_val2[.@i], .@process_opt_val3[.@i], .@process_opt_val4[.@i], .@process_opt_val5[.@i];
			setarray .@opt_param[0],
				.@process_opt_param1[.@i], .@process_opt_param2[.@i], .@process_opt_param3[.@i], .@process_opt_param4[.@i], .@process_opt_param5[.@i];

			.@has_options = 0;
			for (.@opt = 0; .@opt < 5; .@opt++) {
				if (.@opt_id[.@opt]) {
					.@has_options = 1;
					break;
				}
			}

			if (.@process_bound[.@i] > 0) {
				if (.@has_options) {
					getitembound3 .@selected_item_id, 1, .@process_identify[.@i], .@current_refine, .@process_attribute[.@i], .@process_card1[.@i], .@process_card2[.@i], .@process_card3[.@i], .@process_card4[.@i], .@process_bound[.@i], .@opt_id, .@opt_value, .@opt_param;
				} else {
					getitembound2 .@selected_item_id, 1, .@process_identify[.@i], .@current_refine, .@process_attribute[.@i], .@process_card1[.@i], .@process_card2[.@i], .@process_card3[.@i], .@process_card4[.@i], .@process_bound[.@i];
				}
			} else {
				if (.@has_options) {
					getitem3 .@selected_item_id, 1, .@process_identify[.@i], .@current_refine, .@process_attribute[.@i], .@process_card1[.@i], .@process_card2[.@i], .@process_card3[.@i], .@process_card4[.@i], .@opt_id, .@opt_value, .@opt_param;
				} else {
					getitem2 .@selected_item_id, 1, .@process_identify[.@i], .@current_refine, .@process_attribute[.@i], .@process_card1[.@i], .@process_card2[.@i], .@process_card3[.@i], .@process_card4[.@i];
				}
			}

			.@returned_items++;
			dispbottom "  -> Final result: " + .@selected_name$ + " [+" + .@current_refine + "] returned";
			if (.@current_refine > .@process_refine[.@i])
				specialeffect 154,AREA,strnpcinfo(0);
		} else {
			dispbottom "  -> Item destroyed and not returned";
			specialeffect 155,AREA,strnpcinfo(0);
		}

		if (.@stopped_early) {
			.@stopped_items++;
			.@stopped_items += (.@process_count - (.@i + 1));
			break;
		}
	}
	freeloop(0);

	dispbottom "=== Refiner Pro Max complete ===";

	mes "[Bulk Refiner]";
	mes "^00AA00Refining complete.^000000";
	mes "Items selected for processing: " + .@process_count;
	mes "Items actually removed: " + .@processed_items;
	mes "Items returned: " + .@returned_items;
	mes "Items broken: ^FF0000" + .@broken_items + "^000000";
	mes "Successful refine steps: ^00AA00" + .@successful_refines + "^000000";
	mes "Failed refine steps: ^FFAA00" + .@failed_refines + "^000000";
	if (.@missing_items > 0)
		mes "Entries skipped because they changed: ^FF8800" + .@missing_items + "^000000";
	if (.@stopped_items > 0)
		mes "Items left untouched or stopped early: ^808080" + .@stopped_items + "^000000";
	mes .@material_name$ + " used: ^FF6600" + .@total_materials_used + "^000000";
	mes "Zeny spent: ^FF6600" + .@total_zeny_spent + "^000000";
	close;

L_GetSuccessRate:
	.@item_type = getarg(0);
	.@weapon_level = getarg(1);
	.@current_refine = getarg(2);

	if (.@item_type == IT_WEAPON) {
		if (.@weapon_level == 1) {
			if (.@current_refine <= 6) return 100;
			if (.@current_refine == 7) return 60;
			if (.@current_refine == 8) return 40;
			if (.@current_refine == 9) return 20;
		} else if (.@weapon_level == 2) {
			if (.@current_refine <= 5) return 100;
			if (.@current_refine == 6) return 60;
			if (.@current_refine == 7) return 40;
			if (.@current_refine == 8) return 20;
			if (.@current_refine == 9) return 20;
		} else if (.@weapon_level == 3) {
			if (.@current_refine <= 4) return 100;
			if (.@current_refine == 5) return 60;
			if (.@current_refine == 6) return 50;
			if (.@current_refine == 7) return 20;
			if (.@current_refine == 8) return 20;
			if (.@current_refine == 9) return 20;
		} else {
			if (.@current_refine <= 3) return 100;
			if (.@current_refine == 4) return 60;
			if (.@current_refine == 5) return 40;
			if (.@current_refine == 6) return 40;
			if (.@current_refine == 7) return 20;
			if (.@current_refine == 8) return 20;
			if (.@current_refine == 9) return 10;
		}
	} else if (.@item_type == IT_ARMOR) {
		if (.@current_refine <= 3) return 100;
		if (.@current_refine == 4) return 60;
		if (.@current_refine == 5) return 40;
		if (.@current_refine == 6) return 40;
		if (.@current_refine == 7) return 20;
		if (.@current_refine == 8) return 20;
		if (.@current_refine == 9) return 10;
	}

	return 0;
}
