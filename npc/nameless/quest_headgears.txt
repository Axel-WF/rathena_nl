function	script	F_RegisterHeadgearRecipe	{
	.@hat_id = getarg(0);
	.@arg_count = getargcount();

	if (.@hat_id <= 0 || .@arg_count < 3 || ((.@arg_count - 1) % 2) != 0) {
		debugmes "F_RegisterHeadgearRecipe: Invalid recipe declaration for hat ID " + .@hat_id + ".";
		return;
	}

	.@recipe_count = (.@arg_count - 1) / 2;
	for (set .@i, 1; .@i < .@arg_count; set .@i, .@i + 2) {
		.@ingredient_id = getarg(.@i);
		.@ingredient_amount = getarg(.@i + 1);

		if (.@ingredient_id <= 0 || .@ingredient_amount < 1) {
			debugmes "F_RegisterHeadgearRecipe: Invalid ingredient for hat ID " + .@hat_id + ".";
			return;
		}
	}

	setd "$@headgear_recipe_count_" + .@hat_id, .@recipe_count;
	for (set .@i, 1; .@i < .@arg_count; set .@i, .@i + 2) {
		.@slot = (.@i - 1) / 2;
		setd "$@headgear_recipe_item_" + .@hat_id + "[" + .@slot + "]", getarg(.@i);
		setd "$@headgear_recipe_amount_" + .@hat_id + "[" + .@slot + "]", getarg(.@i + 1);
	}
	return;
}

			end;

		.@previous = @craftrecipelog_have[.@i];

			.@changed = 1;
		}
	}

		dispbottom "[Craft Log] " + getitemname(@craftrecipelog_hat) + " is now craftable if you still meet all requirements.";

	addtimer .LogInterval, strnpcinfo(3) + "::OnCraftRecipeLogTick";
	end;

	bindatcmd("craftrecipelog", strnpcinfo(3) + "::OnCraftRecipeLogCommand", 0, 99);
	end;
